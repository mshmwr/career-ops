90 分鐘作答策略（先讀這個）

  1. 開場 3 分鐘把所有題目掃過一遍，先做最有把握、分數高的。HackerRank 是按通過的 test case 給分，不是全有全無。
  2. 先寫能過的暴力解，跑過再優化。卡住別硬撐——單題超過 25 分就跳下一題。
  3. 邊界一定測：空陣列、單一元素、重複值、負數、超大輸入（隱藏 test case 常考效能 → 用 Map/Set 不要 nested loop）。
  4. 讀清楚回傳格式：回 array 還是 string？index 還是值？順序有沒有要求？格式錯整題零分。
  5. 有 console / 自訂測資的話，先用範例 input 驗一次再 submit。
⚠️ 英文題目。看到 "given an array... return..." 先確認回傳型別與順序，再動手。

必背手寫函式（FE HackerRank 最高頻）

這些幾乎每次都出其中幾個。練到能不查資料 90 秒內寫完。

debounce

最後一次呼叫後等 delay 才執行。搜尋框、resize。

function debounce(fn, delay) {
  let t;
  return function (...args) {
    clearTimeout(t);
    t = setTimeout(() => fn.apply(this, args), delay);
  };
}

throttle

每 limit 毫秒最多執行一次。scroll、滑鼠移動。

function throttle(fn, limit) {
  let inThrottle = false;
  return function (...args) {
    if (inThrottle) return;
    fn.apply(this, args);
    inThrottle = true;
    setTimeout(() => (inThrottle = false), limit);
  };
}

deepClone

深拷貝。面試要手寫，別只講 structuredClone。

function deepClone(value) {
  if (value === null || typeof value !== 'object') return value;
  if (value instanceof Date) return new Date(value);
  if (Array.isArray(value)) return value.map(deepClone);
  return Object.fromEntries(
    Object.entries(value).map(([k, v]) => [k, deepClone(v)])
  );
}

flatten（巢狀陣列攤平）

function flatten(arr) {
  return arr.reduce(
    (acc, item) =>
      acc.concat(Array.isArray(item) ? flatten(item) : item),
    []
  );
}

Promise.all polyfill

按順序收集結果，任一 reject 就整體 reject。

function promiseAll(promises) {
  return new Promise((resolve, reject) => {
    const results = [];
    let done = 0;
    if (promises.length === 0) return resolve(results);
    promises.forEach((p, i) => {
      Promise.resolve(p).then((val) => {
        results[i] = val;
        if (++done === promises.length) resolve(results);
      }, reject);
    });
  });
}

curry

function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) return fn.apply(this, args);
    return (...next) => curried.apply(this, [...args, ...next]);
  };
}

memoize

function memoize(fn) {
  const cache = new Map();
  return function (...args) {
    const key = JSON.stringify(args);
    if (cache.has(key)) return cache.get(key);
    const result = fn.apply(this, args);
    cache.set(key, result);
    return result;
  };
}

EventEmitter（on / off / emit）

class EventEmitter {
  constructor() { this.events = new Map(); }
  on(name, fn) {
    if (!this.events.has(name)) this.events.set(name, new Set());
    this.events.get(name).add(fn);
    return this;
  }
  off(name, fn) {
    this.events.get(name)?.delete(fn);
    return this;
  }
  emit(name, ...args) {
    this.events.get(name)?.forEach((fn) => fn(...args));
    return this;
  }
}

演算法核心肌群（用 Map / Set 換時間）

Two Sum — 一次掃描 + hashmap O(n)

function twoSum(nums, target) {
  const seen = new Map();
  for (let i = 0; i < nums.length; i++) {
    const need = target - nums[i];
    if (seen.has(need)) return [seen.get(need), i];
    seen.set(nums[i], i);
  }
  return [];
}

字頻 / anagram

function isAnagram(a, b) {
  if (a.length !== b.length) return false;
  const count = {};
  for (const ch of a) count[ch] = (count[ch] || 0) + 1;
  for (const ch of b) {
    if (!count[ch]) return false;
    count[ch]--;
  }
  return true;
}

常用 one-liner（要熟到反射）

  • 去重：[...new Set(arr)]
  • 排序數字：arr.sort((a, b) => a - b)（預設是字串排序，會錯）
  • 分組計數：arr.reduce((m, x) => (m[x] = (m[x]||0)+1, m), {})
  • 最大值：Math.max(...arr) / 總和：arr.reduce((a,b)=>a+b, 0)
  • chunk：Array.from({length: Math.ceil(a.length/n)}, (_, i) => a.slice(i*n, i*n+n))
  • 陣列：two-sum、去重、分組、滑動視窗、排序比較器
  • 字串：palindrome、anagram、字頻、reverse words
  • 遞迴：flatten、費氏、走訪巢狀物件
  • async：Promise 鏈、async/await 錯誤處理、event loop 順序（micro vs macro task）

React / 前端實作題（若出現）

fetch + 渲染清單（loading / error 三態）

最常見的 React 題型。記得 key、三態、cleanup。

function UserList() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    let alive = true;
    fetch('/api/users')
      .then((r) => {
        if (!r.ok) throw new Error('request failed');
        return r.json();
      })
      .then((data) => alive && setUsers(data))
      .catch((e) => alive && setError(e.message))
      .finally(() => alive && setLoading(false));
    return () => { alive = false; };
  }, []);

  if (loading) return <p>Loading…</p>;
  if (error) return <p>Error: {error}</p>;
  return (
    <ul>
      {users.map((u) => (
        <li key={u.id}>{u.name}</li>
      ))}
    </ul>
  );
}

受控輸入 + 即時過濾

function SearchableList({ items }) {
  const [q, setQ] = useState('');
  const filtered = items.filter((it) =>
    it.toLowerCase().includes(q.toLowerCase())
  );
  return (
    <>
      <input value={q} onChange={(e) => setQ(e.target.value)} />
      <ul>{filtered.map((it) => <li key={it}>{it}</li>)}</ul>
    </>
  );
}
